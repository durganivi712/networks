file_server.c

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <pthread.h>
#include <arpa/inet.h>

#define PORT 8080

void *client_function(void *arg)
{
    int client = *(int *)arg;
    char buffer[1024];
    FILE *fp;
    int n;

    char menu[] =
        "\n--- FILE SHARING SERVER ---\n"
        "1. List Files\n"
        "2. Download File\n"
        "3. Exit\n"
        "Enter choice: ";

    while (1)
    {
        send(client, menu, strlen(menu), 0);

        n = recv(client, buffer, sizeof(buffer) - 1, 0);

        if (n <= 0)
            break;

        buffer[n] = '\0';

        if (strcmp(buffer, "1") == 0)
        {
            char files[] =
                "\nAvailable Files:\n"
                "1. file1.txt\n"
                "2. file2.txt\n"
                "3. notes.txt\n";

            send(client, files, strlen(files), 0);
        }

        else if (strcmp(buffer, "2") == 0)
        {
            send(client, "Enter file name: ", 17, 0);

            n = recv(client, buffer, sizeof(buffer) - 1, 0);

            if (n <= 0)
                break;

            buffer[n] = '\0';

            buffer[strcspn(buffer, "\n")] = '\0';

            fp = fopen(buffer, "r");

            if (fp == NULL)
            {
                send(client, "File not found.", 15, 0);
            }
            else
            {
                while (fgets(buffer, sizeof(buffer), fp) != NULL)
                {
                    send(client, buffer, strlen(buffer), 0);
                }

                fclose(fp);

                send(client, "\n--- FILE END ---", 17, 0);
            }
        }

        else if (strcmp(buffer, "3") == 0)
        {
            break;
        }

        else
        {
            send(client, "Invalid choice.", 15, 0);
        }
    }

    close(client);

    printf("Client disconnected.\n");

    pthread_exit(NULL);
}

int main()
{
    int server, client;
    struct sockaddr_in server_addr, client_addr;
    socklen_t size;

    pthread_t thread;

    server = socket(AF_INET, SOCK_STREAM, 0);

    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(PORT);
    server_addr.sin_addr.s_addr = INADDR_ANY;

    bind(server,
         (struct sockaddr *)&server_addr,
         sizeof(server_addr));

    listen(server, 5);

    printf("File Server waiting...\n");

    while (1)
    {
        size = sizeof(client_addr);

        client = accept(server,
                        (struct sockaddr *)&client_addr,
                        &size);

        printf("Client connected.\n");

        pthread_create(&thread,
                       NULL,
                       client_function,
                       &client);

        pthread_detach(thread);
    }

    close(server);

    return 0;
}

{
    int client;
    struct sockaddr_in server_addr;

    char buffer[1024];
    int n;

    client = socket(AF_INET, SOCK_STREAM, 0);

    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(PORT);
    server_addr.sin_addr.s_addr = inet_addr("127.0.0.1");

    connect(client,
            (struct sockaddr *)&server_addr,
            sizeof(server_addr));

    while (1)
    {
        n = recv(client, buffer, sizeof(buffer) - 1, 0);

        if (n <= 0)
            break;

        buffer[n] = '\0';

        printf("%s", buffer);

        char choice[10];

        scanf("%s", choice);

        send(client, choice, strlen(choice), 0);

        if (strcmp(choice, "1") == 0)
        {
            n = recv(client, buffer, sizeof(buffer) - 1, 0);

            buffer[n] = '\0';

            printf("%s\n", buffer);
        }

        else if (strcmp(choice, "2") == 0)
        {
            n = recv(client, buffer, sizeof(buffer) - 1, 0);

            buffer[n] = '\0';

            printf("%s", buffer);

            scanf("%s", buffer);

            send(client, buffer, strlen(buffer), 0);

            printf("\nFile Content:\n");

            while (1)
            {
                n = recv(client, buffer, sizeof(buffer) - 1, 0);

                if (n <= 0)
                    break;

                buffer[n] = '\0';

                printf("%s", buffer);

                if (strstr(buffer, "FILE END") != NULL)
                    break;

                if (strlen(buffer) < 1023)
                    break;
            }
        }

        else if (strcmp(choice, "3") == 0)
        {
            break;
        }
    }

    close(client);

    return 0;
}




chat_server.c

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <pthread.h>
#include <arpa/inet.h>

#define PORT 8081
#define MAX_CLIENTS 10

int clients[MAX_CLIENTS];
int count = 0;

pthread_mutex_t lock = PTHREAD_MUTEX_INITIALIZER;

void broadcast_message(char *message, int sender)
{
    pthread_mutex_lock(&lock);

    for (int i = 0; i < count; i++)
    {
        if (clients[i] != sender)
        {
            send(clients[i],
                 message,
                 strlen(message),
                 0);
        }
    }

    pthread_mutex_unlock(&lock);
}

void remove_client(int client)
{
    pthread_mutex_lock(&lock);

    for (int i = 0; i < count; i++)
    {
        if (clients[i] == client)
        {
            for (int j = i; j < count - 1; j++)
            {
                clients[j] = clients[j + 1];
            }

            count--;
            break;
        }
    }

    pthread_mutex_unlock(&lock);
}

void *client_function(void *arg)
{
    int client = *(int *)arg;
    char message[1024];

    send(client,
         "Connected to Chat Server!\n",
         27,
         0);

    while (1)
    {
        int n = recv(client,
                     message,
                     sizeof(message) - 1,
                     0);

        if (n <= 0)
            break;

        message[n] = '\0';

        if (strcmp(message, "exit") == 0)
            break;

        broadcast_message(message, client);
    }

    remove_client(client);

    close(client);

    printf("Client disconnected.\n");

    pthread_exit(NULL);
}

int main()
{
    int server, client;
    struct sockaddr_in server_addr, client_addr;
    socklen_t size;

    pthread_t thread;

    server = socket(AF_INET, SOCK_STREAM, 0);

    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(PORT);
    server_addr.sin_addr.s_addr = INADDR_ANY;

    bind(server,
         (struct sockaddr *)&server_addr,
         sizeof(server_addr));

    listen(server, 10);

    printf("Chat Server waiting...\n");

    while (1)
    {
        size = sizeof(client_addr);

        client = accept(server,
                        (struct sockaddr *)&client_addr,
                        &size);

        pthread_mutex_lock(&lock);

        if (count < MAX_CLIENTS)
        {
            clients[count] = client;
            count++;
        }

        pthread_mutex_unlock(&lock);

        printf("New client connected.\n");

        pthread_create(&thread,
                       NULL,
                       client_function,
                       &client);

        pthread_detach(thread);
    }

    close(server);

    return 0;
}

int client;

void *receive_messages(void *arg)
{
    char message[1024];
    int n;

    while (1)
    {
        n = recv(client,
                 message,
                 sizeof(message) - 1,
                 0);

        if (n <= 0)
            break;

        message[n] = '\0';

        printf("\nMessage: %s\n", message);
        printf("Enter message: ");
        fflush(stdout);
    }

    return NULL;
}

int main()
{
    struct sockaddr_in server_addr;

    pthread_t thread;

    char message[1024];

    client = socket(AF_INET, SOCK_STREAM, 0);

    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(PORT);
    server_addr.sin_addr.s_addr =
        inet_addr("127.0.0.1");

    connect(client,
            (struct sockaddr *)&server_addr,
            sizeof(server_addr));

    pthread_create(&thread,
                   NULL,
                   receive_messages,
                   NULL);

    while (1)
    {
        printf("Enter message: ");

        scanf(" %[^\n]", message);

        send(client,
             message,
             strlen(message),
             0);

        if (strcmp(message, "exit") == 0)

    }

    return 0;
}
